src/
├── main.py                                    # Application entry point; orchestrates the full pipeline
├── youtube/                                   # YouTube transcript fetching domain
│   ├── __init__.py
│   └── transcript_fetcher.py                 # Fetches transcripts via YouTube API or falls back to audio transcription
├── prompts/                                   # Prompt system (prompt metadata, template, composition, factory)
│   ├── __init__.py
│   ├── prompt_metadata.py                    # PromptMetadata dataclass: name, description, parameters
│   ├── prompt_template.py                    # PromptTemplate dataclass: role, content with placeholder rendering
│   ├── prompt.py                             # Prompt dataclass: composes metadata + list of templates
│   └── prompt_factory.py                     # PromptFactory: loads prompt YAML files, resolves placeholders
├── processors/                                # Core processing pipeline (translate + validate)
│   ├── __init__.py
│   ├── processor.py                          # Processor: orchestrates translation & validation via LLM
│   ├── processed_script.py                   # ProcessedScript dataclass: original/translated script + validation results
│   ├── translated_segment.py                 # TranslatedSegment (Pydantic): LLM output schema for translation
│   └── translation_validation.py             # TranslationValidation (Pydantic): LLM output schema for validation
├── pipeline/                                  # Cross-cutting pipeline infrastructure
│   ├── __init__.py
│   └── statistics/
│       ├── __init__.py
│       ├── execution.py                      # @execution_time decorator & track_execution context manager
│       ├── timing_node.py                    # TimingNode dataclass: tree node for nested timing
│       └── timing_tracker.py                 # TimingTracker: manages timing stack & exports statistics
├── llm/                                       # LLM provider abstractions
│   ├── __init__.py
│   ├── llm_provider.py                       # LLMProvider (ABC): base class with retry/validation logic
│   ├── open_code_provider.py                 # OpenCodeProvider: local OpenCode CLI wrapper
│   └── provider_models.py                    # ProviderModels enum: OLLAMA, OPEN_CODE
├── transcribers/                              # Audio transcription domain
│   ├── __init__.py
│   ├── transcriber.py                        # Transcriber (ABC): downloads audio, loads model, transcribes
│   ├── segment.py                            # process_segment(): formats Whisper segments for output
│   └── languages/
│       ├── __init__.py
│       ├── english_transcriber.py            # EnglishTranscriber: uses openai-whisper
│       ├── bengali_transcriber.py            # BengaliTranscriber: uses banglaspeech2text
│       └── hindi_transcriber.py              # HindiTranscriber: uses faster-whisper
└── utils/                                     # Shared utilities (grouped by concern)
    ├── __init__.py
    ├── config/                                # Configuration objects & loading
    │   ├── __init__.py
    │   ├── youtube_config.py                 # YoutubeConfig: URL, language, transcriber instance
    │   ├── models_config.py                  # ModelsConfig: translator/validator model names
    │   ├── llm_config.py                     # LLMConfig: provider, models, instantiated provider
    │   ├── app_config.py                     # AppConfig: root config (youtube, output_dir, llm)
    │   ├── app_config_service.py             # AppConfigService: loads & parses config.yaml
    │   └── dependency_container.py           # DependencyContainer: wires FileSystem, EnvironmentSystem, PromptFactory
    ├── io/                                    # I/O operations
    │   ├── __init__.py
    │   ├── filesystem.py                     # FileSystem: path ops, YAML read/write, dir creation
    │   ├── audio_downloader.py               # download_audio_as_wav(): yt-dlp + ffmpeg audio extraction
    │   ├── environment_system.py             # EnvironmentSystem: executable lookup, subprocess, Ollama model mgmt
    │   └── video.py                          # extract_video_id(): parses YouTube URLs for video ID
    ├── validation/                            # Validation helpers
    │   ├── __init__.py
    │   ├── validators.py                     # require_field(), require_directory_path(), validate_parameters()
    │   └── supported_languages.py            # SupportedLanguages enum: ENGLISH, HINDI, BENGALI + codes
    ├── naming/                                # Dynamic class resolution
    │   ├── __init__.py
    │   ├── resolver.py                       # get_llm_provider_class_name(), get_transcriber()
    │   └── naming_case.py                    # snake_to_pascal_case()
    └── formatting/                            # String constants & formatters
        ├── __init__.py
        ├── strings.py                        # All string constants (keys, messages, patterns, paths)
        └── formatters.py                     # format_time(): seconds -> "Xh Ym Zs"
